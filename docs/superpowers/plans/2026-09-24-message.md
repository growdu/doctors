# 站内信 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 解决 `docs/REVIEW-REPORT.md` 中"站内信评审 P2"：当前 `services/message/internal/service/` 仅有 8 个占位单测（基于 `orderID` 直发消息），缺 DB 表、仓储、会话创建、消息历史、鉴权、handler、路由、main。产出患者 / 陪诊师可在订单上下文里收发消息的全链路，按 `2026-09-24-l2-api-gap-design.md` §2.1 P2 + §3.1 `Conversation` / `Message` entity 实现 4 个 REST API。v1 不做实时推送（v2 引 WebSocket）；前端 5s 轮询。

## Global Constraints

- Go 1.24+（toolchain go1.24.3）
- pgx v5.7.1（v1 不引 ORM，全部 pgx 直写 SQL）
- 测试覆盖率：业务包 ≥ 80%
- Commit 节奏：每个 Task 完成立即 commit；前缀 `feat:` / `test:` / `fix:` / `docs:`
- 所有响应走 `shared/httpx`（业务码在 body）
- 错误统一 `shared/errs.Error`（业务码 5 位 / 系统码 6 位）
- 一个 conversation 关联一个 order：`UNIQUE(conversations.order_id)`；v1 简化，不支持同一订单多个会话
- participants 锁定为 `[patient_id, escort_id]` 两位（来自 orders.patient_id / orders.escort_id），创建会话时确定，后续不能改
- `messages.content_type` 枚举：`'text'` / `'image'`（v1 不支持 `'location'` / `'voice'` 等）
- `messages.content`：`text` 直接存文本（1..1000 字符）；`image` 存 base64 字符串（≤ 200KB base64 ≈ 150KB 二进制；v1 不接 OSS）
- v1 不在 DB 维护 unread count；前端用 `conversations.last_message_at` vs 本地最后拉取时间判定未读
- v1 不做 WebSocket 实时推送；前端轮询 5s（前端控制，后端只负责 GET 拉新）
- v1 不做"已读 / 未读"持久化；前端会话进入即视为已读
- 鉴权：JWT user_id；只能操作自己作为 participant 的会话（patient_id 或 escort_id）；非参与者返回 `CodeForbidden`
- 创建会话幂等：`(order_id, user_id, peer_user_id)` 三元组若已存在，直接返回已存在会话（避免重复会话）

```sql
-- 0007_messages.up.sql
-- 站内信：会话（conversations）+ 消息（messages）。
-- 设计要点：
--   - 一个 conversation 关联一个 order：UNIQUE(order_id)；v1 简化，不支持同订单多会话。
--   - participants 锁定为 [patient_id, escort_id]，创建时确定，后续不能改。
--   - last_message_at 是读会话列表时的"排序键 + 未读指示"（v1 不存 unread count）。
--   - content_type 枚举：'text' / 'image'（v1 不支持 location / voice）。
--   - v1 不做"已读 / 未读"持久化；前端进入会话即视为已读。

CREATE TABLE conversations (
  id BIGSERIAL PRIMARY KEY,
  order_id BIGINT NOT NULL REFERENCES orders(id),
  patient_id BIGINT NOT NULL REFERENCES users(id),
  escort_id BIGINT NOT NULL REFERENCES users(id),
  last_message_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  last_message_preview TEXT,
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- 一个订单最多一个会话；重复创建走 ON CONFLICT 幂等返回。
CREATE UNIQUE INDEX uq_conversations_order ON conversations(order_id);

-- 患者 / 陪诊师拉"我的会话"（GET /api/v1/messages/conversations）。
CREATE INDEX idx_conversations_patient ON conversations(patient_id, last_message_at DESC);
CREATE INDEX idx_conversations_escort ON conversations(escort_id, last_message_at DESC);

CREATE TABLE messages (
  id BIGSERIAL PRIMARY KEY,
  conversation_id BIGINT NOT NULL REFERENCES conversations(id),
  sender_id BIGINT NOT NULL REFERENCES users(id),
  content_type VARCHAR(16) NOT NULL CHECK (content_type IN ('text','image')),
  content TEXT NOT NULL,
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- 拉某会话的消息历史（GET /api/v1/messages/conversations/:id/messages）。
CREATE INDEX idx_messages_conversation_created ON messages(conversation_id, created_at DESC);
```

```go
// Package repo 是 message-service 的数据访问层。
//
// 设计要点：
//   - 用 pgx 直写 SQL（不引 sqlc）。
//   - CreateConversation 幂等：UNIQUE(order_id) 冲突时返回原 ID（ON CONFLICT DO NOTHING + 回查）。
//   - ListMessages 按 created_at DESC + LIMIT 翻页；ListMessagesBefore 用于"加载更早历史"。
//   - UpdateLastMessage 在 PostMessage 后调用，让会话列表展示新消息预览。
package repo

import (
	"context"
	"errors"
	"fmt"
	"time"

	"github.com/jackc/pgx/v5"
	"github.com/jackc/pgx/v5/pgxpool"
)

// Conversation 映射 conversations 表行。
type Conversation struct {
	ID                 int64
	OrderID            int64
	PatientID          int64
	EscortID           int64
	LastMessageAt      time.Time
	LastMessagePreview string
	CreatedAt          time.Time
	UpdatedAt          time.Time
}

// Message 映射 messages 表行。
type Message struct {
	ID             int64
	ConversationID int64
	SenderID       int64
	ContentType    string // "text" | "image"
	Content        string
	CreatedAt      time.Time
}

// ErrConversationNotFound 是查无结果的哨兵。
var ErrConversationNotFound = errors.New("repo: conversation not found")

// MessageRepo 是 conversations + messages 表的仓储。
type MessageRepo struct {
	pool *pgxpool.Pool
}

// NewMessageRepo 构造仓储。
func NewMessageRepo(pool *pgxpool.Pool) *MessageRepo { return &MessageRepo{pool: pool} }

// CreateConversation 幂等创建会话（同 order_id 重复调用返回原 ID）。
func (r *MessageRepo) CreateConversation(ctx context.Context, c *Conversation) error {
	const q = `
		INSERT INTO conversations (order_id, patient_id, escort_id)
		VALUES ($1, $2, $3)
		ON CONFLICT (order_id) DO NOTHING
		RETURNING id, created_at, updated_at, last_message_at`
	err := r.pool.QueryRow(ctx, q, c.OrderID, c.PatientID, c.EscortID).
		Scan(&c.ID, &c.CreatedAt, &c.UpdatedAt, &c.LastMessageAt)
	if errors.Is(err, pgx.ErrNoRows) {
		// 冲突：回查原 ID
		return r.pool.QueryRow(ctx,
			`SELECT id, created_at, updated_at, last_message_at FROM conversations WHERE order_id = $1`,
			c.OrderID,
		).Scan(&c.ID, &c.CreatedAt, &c.UpdatedAt, &c.LastMessageAt)
	}
	return err
}

// GetConversation 按 ID 查会话；查无返回 (nil, ErrConversationNotFound)。
func (r *MessageRepo) GetConversation(ctx context.Context, id int64) (*Conversation, error) {
	const q = `
		SELECT id, order_id, patient_id, escort_id,
		       last_message_at, COALESCE(last_message_preview, ''),
		       created_at, updated_at
		FROM conversations WHERE id = $1`
	c := &Conversation{}
	err := r.pool.QueryRow(ctx, q, id).Scan(
		&c.ID, &c.OrderID, &c.PatientID, &c.EscortID,
		&c.LastMessageAt, &c.LastMessagePreview,
		&c.CreatedAt, &c.UpdatedAt,
	)
	if err != nil {
		if errors.Is(err, pgx.ErrNoRows) {
			return nil, ErrConversationNotFound
		}
		return nil, fmt.Errorf("get conversation: %w", err)
	}
	return c, nil
}

// ListConversationsByUser 按用户拉会话（patient_id 或 escort_id 匹配）；按 last_message_at DESC。
func (r *MessageRepo) ListConversationsByUser(ctx context.Context, userID int64, limit, offset int) ([]*Conversation, error) {
	const q = `
		SELECT id, order_id, patient_id, escort_id,
		       last_message_at, COALESCE(last_message_preview, ''),
		       created_at, updated_at
		FROM conversations
		WHERE patient_id = $1 OR escort_id = $1
		ORDER BY last_message_at DESC
		LIMIT $2 OFFSET $3`
	rows, err := r.pool.Query(ctx, q, userID, limit, offset)
	if err != nil {
		return nil, fmt.Errorf("list conversations: %w", err)
	}
	defer rows.Close()
	out := make([]*Conversation, 0, limit)
	for rows.Next() {
		c := &Conversation{}
		if err := rows.Scan(
			&c.ID, &c.OrderID, &c.PatientID, &c.EscortID,
			&c.LastMessageAt, &c.LastMessagePreview,
			&c.CreatedAt, &c.UpdatedAt,
		); err != nil {
			return nil, err
		}
		out = append(out, c)
	}
	return out, rows.Err()
}

// CreateMessage 插入一条消息；ID + CreatedAt 由 DB 回写。
// content_type CHECK 由 DB 校验（失败返回 error）。
func (r *MessageRepo) CreateMessage(ctx context.Context, m *Message) error {
	const q = `
		INSERT INTO messages (conversation_id, sender_id, content_type, content)
		VALUES ($1, $2, $3, $4)
		RETURNING id, created_at`
	return r.pool.QueryRow(ctx, q, m.ConversationID, m.SenderID, m.ContentType, m.Content).
		Scan(&m.ID, &m.CreatedAt)
}

// ListMessages 拉某会话的最新 N 条（按 created_at DESC 翻页）。
func (r *MessageRepo) ListMessages(ctx context.Context, conversationID int64, limit, offset int) ([]*Message, error) {
	const q = `
		SELECT id, conversation_id, sender_id, content_type, content, created_at
		FROM messages
		WHERE conversation_id = $1
		ORDER BY created_at DESC
		LIMIT $2 OFFSET $3`
	rows, err := r.pool.Query(ctx, q, conversationID, limit, offset)
	if err != nil {
		return nil, fmt.Errorf("list messages: %w", err)
	}
	defer rows.Close()
	out := make([]*Message, 0, limit)
	for rows.Next() {
		m := &Message{}
		if err := rows.Scan(&m.ID, &m.ConversationID, &m.SenderID, &m.ContentType, &m.Content, &m.CreatedAt); err != nil {
			return nil, err
		}
		out = append(out, m)
	}
	return out, rows.Err()
}

// ListMessagesBefore 拉早于 beforeID 的 N 条（向上翻页）。
func (r *MessageRepo) ListMessagesBefore(ctx context.Context, conversationID, beforeID int64, limit int) ([]*Message, error) {
	const q = `
		SELECT id, conversation_id, sender_id, content_type, content, created_at
		FROM messages
		WHERE conversation_id = $1 AND id < $2
		ORDER BY created_at DESC
		LIMIT $3`
	rows, err := r.pool.Query(ctx, q, conversationID, beforeID, limit)
	if err != nil {
		return nil, fmt.Errorf("list messages before: %w", err)
	}
	defer rows.Close()
	out := make([]*Message, 0, limit)
	for rows.Next() {
		m := &Message{}
		if err := rows.Scan(&m.ID, &m.ConversationID, &m.SenderID, &m.ContentType, &m.Content, &m.CreatedAt); err != nil {
			return nil, err
		}
		out = append(out, m)
	}
	return out, rows.Err()
}

// UpdateLastMessage 更新会话的 last_message_at + last_message_preview（PostMessage 后调用）。
func (r *MessageRepo) UpdateLastMessage(ctx context.Context, conversationID int64, at time.Time, preview string) error {
	const q = `
		UPDATE conversations
		   SET last_message_at = $2,
		       last_message_preview = $3,
		       updated_at = NOW()
		 WHERE id = $1`
	_, err := r.pool.Exec(ctx, q, conversationID, at, preview)
	if err != nil {
		return fmt.Errorf("update last message: %w", err)
	}
	return nil
}
```

```go
package events

import (
	"context"
	"encoding/json"
	"errors"
	"fmt"
	"strconv"
	"time"

	"github.com/segmentio/kafka-go"

	"github.com/growdu/doctors/shared/contracts"

// Publisher 抽象消息事件发布。
type Publisher interface {
	PublishMessageSent(ctx context.Context, ev contracts.MessageSentEvent) error
	Close() error
}

// KafkaPublisher 用 kafka-go writer 写 topic。
type KafkaPublisher struct {
	writer *kafka.Writer
}

// NewKafkaPublisher 构造 publisher。
func NewKafkaPublisher(brokers []string) *KafkaPublisher {
	return &KafkaPublisher{
		writer: &kafka.Writer{
			Addr:         kafka.TCP(brokers...),
			Balancer:     &kafka.LeastBytes{},
			BatchTimeout: 50 * time.Millisecond,
			RequiredAcks: kafka.RequireOne,
			Async:        false,
		},
	}
}

// Close 关闭 writer。
func (p *KafkaPublisher) Close() error { return p.writer.Close() }

// publish 通用写消息。
func (p *KafkaPublisher) publish(ctx context.Context, topic string, key string, body any) error {
	if p == nil || p.writer == nil {
		return errors.New("publisher: writer is nil")
	}
	data, err := json.Marshal(body)
	if err != nil {
		return fmt.Errorf("marshal %s: %w", topic, err)
	}
	return p.writer.WriteMessages(ctx, kafka.Message{
		Topic: topic, Key: []byte(key), Value: data, Time: time.Now(),
	})
}

// PublishMessageSent 发布 message.sent。
func (p *KafkaPublisher) PublishMessageSent(ctx context.Context, ev contracts.MessageSentEvent) error {
	return p.publish(ctx, contracts.TopicMessageSent, strconv.FormatInt(ev.ConversationID, 10), ev)
}

// NopPublisher 是测试 / dev 占位。

	SentCount int
}

// PublishMessageSent 计数。
func (p *NopPublisher) PublishMessageSent(_ context.Context, _ contracts.MessageSentEvent) error {
	p.SentCount++
	return nil
}

// Close 无资源。
func (p *NopPublisher) Close() error { return nil }
```

```go
// Package service - message-service 业务编排。
//
// 设计要点：
//   - v1 一个 conversation 关联一个 order（UNIQUE(order_id)）；同 order 重复创建幂等返回原会话。
//   - SendMessage 校验 sender ∈ {patient_id, escort_id}，否则 CodeForbidden。
//   - content_type 枚举：'text'（1..1000 字符） / 'image'（base64 ≤ 200KB）；其他类型 CodeParamInvalid。
//   - PostMessage 后异步（best-effort）发布 contracts.MessageSentEvent（带 ConversationID + ContentType）。
//   - v1 不维护 unread_count；前端用 conversations.last_message_at vs 本地时间判定未读。
package service

import (
	"context"
	"encoding/base64"
	"errors"
	"strings"
	"time"
	"unicode/utf8"

	"github.com/growdu/doctors/shared/contracts"
	"github.com/growdu/doctors/shared/errs"
)

// 内容上限常量。
const (
	maxTextRunes  = 1000   // text 最多 1000 字符
	maxImageBytes = 200000 // image base64 <= 200KB（≈ 150KB 二进制）
)

// Conversation 是会话。
type Conversation struct {
	ID                 int64
	OrderID            int64
	PatientID          int64
	EscortID           int64
	LastMessageAt      time.Time
	LastMessagePreview string
	CreatedAt          time.Time
	UpdatedAt          time.Time
}

// Message 是单条消息。
type Message struct {
	ID             int64
	ConversationID int64
	SenderID       int64
	ContentType    string // "text" | "image"
	Content        string
	CreatedAt      time.Time
}

// Repo 是仓储契约。
type Repo interface {
	CreateConversation(ctx context.Context, c *Conversation) error
	GetConversation(ctx context.Context, id int64) (*Conversation, error)
	ListConversationsByUser(ctx context.Context, userID int64, limit, offset int) ([]*Conversation, error)
	CreateMessage(ctx context.Context, m *Message) error
	ListMessages(ctx context.Context, conversationID int64, limit, offset int) ([]*Message, error)
	ListMessagesBefore(ctx context.Context, conversationID, beforeID int64, limit int) ([]*Message, error)
	UpdateLastMessage(ctx context.Context, conversationID int64, at time.Time, preview string) error
}

// Publisher 是事件发布抽象。
type Publisher interface {
	PublishMessageSent(ctx context.Context, ev contracts.MessageSentEvent) error
}

// ErrConversationNotFound 是查无结果。
var ErrConversationNotFound = errors.New("message service: conversation not found")

// Service 是 message 业务编排器。
type Service struct {
	repo      Repo
	publisher Publisher
	clockNow  func() time.Time
}

// New 装配 Service。
func New(r Repo, p Publisher) *Service {
	return &Service{repo: r, publisher: p, clockNow: time.Now}
}

// WithClock 注入时钟（测试用）。
func (s *Service) WithClock(now func() time.Time) *Service {
	s.clockNow = now
	return s
}

// CreateConversation 按 order_id 创建会话；幂等（已存在则返回原会话）。
//   - orderID / userID / peerID 必须 > 0
//   - userID != peerID
func (s *Service) CreateConversation(ctx context.Context, orderID, userID, peerID int64) (*Conversation, error) {
	if orderID == 0 || userID == 0 || peerID == 0 {
		return nil, errs.New(errs.CodeParamInvalid, "order_id / user_id / peer_id required")
	}
	if userID == peerID {
		return nil, errs.New(errs.CodeParamInvalid, "user and peer must differ")
	}
	// v1 简化：caller 必须传 patient_id 作为 userID，escort_id 作为 peerID；
	// 真实从 orders.patient_id / escort_id 拿是后续 plan 的事（避免 service 间循环依赖）。
	// v1 信任前端传参。
	c := &Conversation{OrderID: orderID, PatientID: userID, EscortID: peerID}
	if err := s.repo.CreateConversation(ctx, c); err != nil {
		return nil, errs.Wrap(errs.CodeInternal, "create conversation", err)
	}
	return c, nil
}

// SendMessage 在某会话里发一条消息。
//   - senderID 必须是会话的 patient_id 或 escort_id
//   - contentType ∈ {"text","image"}
//   - text：content 1..1000 字符
//   - image：content 是 base64 字符串，decode 后 ≤ 200KB
func (s *Service) SendMessage(ctx context.Context, conversationID, senderID int64, contentType, content string) (*Message, error) {
	if conversationID == 0 || senderID == 0 {
		return nil, errs.New(errs.CodeParamInvalid, "conversation_id / sender_id required")
	}
	conv, err := s.repo.GetConversation(ctx, conversationID)
	if err != nil {
		if errors.Is(err, ErrConversationNotFound) {
			return nil, errs.New(errs.CodeNotFound, "conversation not found")
		}
		return nil, errs.Wrap(errs.CodeInternal, "get conversation", err)
	}
	if senderID != conv.PatientID && senderID != conv.EscortID {
		return nil, errs.New(errs.CodeForbidden, "sender is not a participant of this conversation")
	}

	// 校验 content_type + content
	switch contentType {
	case "text":
		l := utf8.RuneCountInString(strings.TrimSpace(content))
		if l == 0 || l > maxTextRunes {
			return nil, errs.New(errs.CodeParamInvalid, "text length must be 1..1000")
		}
	case "image":
		decoded, derr := base64.StdEncoding.DecodeString(content)
		if derr != nil || len(decoded) == 0 || len(decoded) > maxImageBytes {
			return nil, errs.New(errs.CodeParamInvalid, "image must be valid base64 (<=200KB)")
		}
	default:
		return nil, errs.New(errs.CodeParamInvalid, "content_type must be 'text' or 'image'")
	}

	m := &Message{
		ConversationID: conversationID,
		SenderID:       senderID,
		ContentType:    contentType,
		Content:        content,
	}
	if err := s.repo.CreateMessage(ctx, m); err != nil {
		return nil, errs.Wrap(errs.CodeInternal, "create message", err)
	}

	// 更新会话 last_message_at + preview
	preview := previewOf(contentType, content)
	now := s.clockNow()
	_ = s.repo.UpdateLastMessage(ctx, conversationID, now, preview)

	// 异步（best-effort）发布事件
	if s.publisher != nil {
		var peerID int64
		if senderID == conv.PatientID {
			peerID = conv.EscortID
		} else {
			peerID = conv.PatientID
		}
		_ = s.publisher.PublishMessageSent(ctx, contracts.MessageSentEvent{
			MessageID:      m.ID,
			ConversationID: conversationID,
			OrderID:        conv.OrderID,
			FromID:         senderID,
			ToID:           peerID,
			ContentType:    contentType,
			Body:           content,
			SentAt:         m.CreatedAt,
		})
	}
	return m, nil
}

// ListConversations 拉我的会话列表（按 last_message_at DESC）。
func (s *Service) ListConversations(ctx context.Context, userID int64, limit, offset int) ([]*Conversation, error) {
	if userID == 0 {
		return nil, errs.New(errs.CodeParamInvalid, "user_id required")
	}
	if limit <= 0 || limit > 100 {
		limit = 50
	}
	return s.repo.ListConversationsByUser(ctx, userID, limit, offset)
}

// ListMessages 拉某会话的最新 N 条消息；viewer 必须是参与者。
func (s *Service) ListMessages(ctx context.Context, conversationID, viewerID int64, limit, offset int) ([]*Message, error) {
	if conversationID == 0 || viewerID == 0 {
		return nil, errs.New(errs.CodeParamInvalid, "conversation_id / viewer_id required")
	}
	conv, err := s.repo.GetConversation(ctx, conversationID)
	if err != nil {
		if errors.Is(err, ErrConversationNotFound) {
			return nil, errs.New(errs.CodeNotFound, "conversation not found")
		}
		return nil, errs.Wrap(errs.CodeInternal, "get conversation", err)
	}
	if viewerID != conv.PatientID && viewerID != conv.EscortID {
		return nil, errs.New(errs.CodeForbidden, "viewer is not a participant of this conversation")
	}
	if limit <= 0 || limit > 100 {
		limit = 50
	}
	return s.repo.ListMessages(ctx, conversationID, limit, offset)
}

// previewOf 构造最后一条消息的预览（text 取前 30 字；image 标 "[图片]"）。
func previewOf(contentType, content string) string {
	if contentType == "image" {
		return "[图片]"
	}
	r := []rune(strings.TrimSpace(content))
	if len(r) > 30 {
		return string(r[:30]) + "..."
	}
	return string(r)
}
```

```go
// Package middleware 提供 message-service 专用鉴权中间件。
//
// 与 order-service / auth-service 的中间件契约一致：Bearer JWT → 解析 claims → 写入 ctx。
package middleware

import (
	"strings"

	"github.com/gin-gonic/gin"

	authpkg "github.com/growdu/doctors/shared/auth"
	"github.com/growdu/doctors/shared/errs"
	"github.com/growdu/doctors/shared/httpx"
)

const (
	// UserIDKey 是 gin.Context 里 user_id 的 key。
	UserIDKey = "message_user_id"
)

// Auth 校验 Bearer JWT；通过则把 user_id 写入 ctx。
func Auth(secret string) gin.HandlerFunc {
	return func(c *gin.Context) {
		h := c.GetHeader("Authorization")
		if !strings.HasPrefix(h, "Bearer ") {
			httpx.Fail(c, int(errs.CodeUnauthorized), "missing bearer token")
			c.Abort()
			return
		}
		tok := strings.TrimPrefix(h, "Bearer ")
		claims, err := authpkg.Parse(secret, tok)
		if err != nil {
			httpx.Fail(c, int(errs.CodeUnauthorized), "invalid token")
			c.Abort()
			return
		}
		c.Set(UserIDKey, claims.UserID)
		c.Next()
	}
}

// UserID 从 ctx 读取 user_id；缺失返回 0。
func UserID(c *gin.Context) int64 {
	v, ok := c.Get(UserIDKey)
	if !ok {
		return 0
	}
	id, _ := v.(int64)
	return id
}
```

```go
// respondError 统一错误翻译。
func respondError(c *gin.Context, err error) {
	if e, ok := errs.As(err); ok {
		httpx.Fail(c, int(e.Code), e.Msg)
		return
	}
	httpx.Fail(c, int(errs.CodeInternal), err.Error())
}

// parseID 把路径 :id 解析为 int64。
func parseID(c *gin.Context) (int64, bool) {
	idStr := c.Param("id")
	id, err := strconv.ParseInt(idStr, 10, 64)
	if err != nil {
		respondError(c, errs.New(errs.CodeParamInvalid, "invalid id"))
		return 0, false
	}
	return id, true
}
```

```go
// Package server 包装 HTTP server，支持优雅停机。
package server

import (
	"context"
	"net/http"
	"time"

	"go.uber.org/zap"

	"github.com/growdu/doctors/shared/logger"
)

// Server 是 HTTP server 包装。
type Server struct {
	addr string
	srv  *http.Server
}

// New 构造 server。
func New(addr string, h http.Handler) *Server {
	return &Server{
		addr: addr,
		srv: &http.Server{
			Addr:              addr,
			Handler:           h,
			ReadHeaderTimeout: 5 * time.Second,
		},
	}
}

// Run 启动 + 优雅停机。
func (s *Server) Run(ctx context.Context) error {
	errCh := make(chan error, 1)
	go func() {
		if err := s.srv.ListenAndServe(); err != nil && err != http.ErrServerClosed {
			errCh <- err
			return
		}
		errCh <- nil
	}()
	logger.L().Info("message-service starting", zap.String("addr", s.addr))

	select {
	case <-ctx.Done():
		shutdownCtx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
		defer cancel()
		return s.srv.Shutdown(shutdownCtx)
	case err := <-errCh:
		return err
	}
}
```

```go
package main

import (

	"github.com/growdu/doctors/shared/logger"
)

func main() {
	cfg, err := config.Load("message")
	if err != nil {
		log.Fatalf("load config: %v", err)
	}
	logger.SetLevel(parseLevel(cfg.Logging.Level))
	defer func() { _ = logger.L().Sync() }()

	ctx, stop := signal.NotifyContext(context.Background(), os.Interrupt, syscall.SIGTERM)
	defer stop()

	pool, err := pgxpool.New(ctx, cfg.Postgres.DSN)
	if err != nil {
		log.Fatalf("connect pg: %v", err)
	}
	defer pool.Close()

	var pub events.Publisher
	if len(cfg.Kafka.Brokers) > 0 {
		kp := events.NewKafkaPublisher(cfg.Kafka.Brokers)
		defer func() { _ = kp.Close() }()
		pub = kp
	} else {
		pub = &events.NopPublisher{}
	}

	r := repo.NewMessageRepo(pool)
	svc := service.New(r, pub)
	h := handler.New(svc)
	srv := server.New(cfg.HTTP.Addr, router.New(h, cfg.JWT.Secret))

	if err := srv.Run(ctx); err != nil {
		log.Fatalf("server run: %v", err)
	}
}

// parseLevel 把字符串日志级别映射到 zapcore.Level。
func parseLevel(s string) zapcore.Level {
	switch s {
	case "debug":
		return zapcore.DebugLevel
	case "warn":
		return zapcore.WarnLevel
	case "error":
		return zapcore.ErrorLevel
	default:
		return zapcore.InfoLevel
	}
}
```
