  resolved_at TIMESTAMPTZ,

// Package repo 是 sos-service 的数据访问层。
//
// 设计要点：
//   - 用 pgx 直写 SQL（不引 sqlc）。
//   - status 用 CHECK 约束（DB 层 latest 已校验）。
//   - LatestByOrder 按 created_at DESC 取最近一条；服务层做 5 分钟去重。
package repo

import (
	"context"
	"errors"
	"fmt"
	"time"

	"github.com/jackc/pgx/v5"
	"github.com/jackc/pgx/v5/pgxpool"
)

// Record 映射 sos_records 表行。
type Record struct {
	ID             int64
	OrderID        int64
	TriggerBy      int64
	TriggerRole    string // "patient" | "escort"
	Lat            *float64
	Lng            *float64
	Address        string
	Status         string // "raised" | "resolved" | "dismissed"
	CreatedAt      time.Time
	ResolvedAt     *time.Time
	ResolvedBy     *int64
	ResolutionNote string
}

// ErrSosNotFound 是查询无结果时的哨兵。
var ErrSosNotFound = errors.New("repo: sos not found")

// SosRepo 是 sos_records 表的仓储。
type SosRepo struct {
	pool *pgxpool.Pool
}

// NewSosRepo 构造仓储。
func NewSosRepo(pool *pgxpool.Pool) *SosRepo { return &SosRepo{pool: pool} }

// baseSelect 是 SELECT 子句。
const baseSelect = `
	SELECT id, order_id, trigger_by, trigger_role, lat, lng, address,
	       status, created_at, resolved_at, resolved_by, resolution_note
	FROM sos_records`

// Create 插入一条 SOS；ID / CreatedAt 由 DB 回写。
func (r *SosRepo) Create(ctx context.Context, s *Record) error {
	const q = `
		INSERT INTO sos_records (order_id, trigger_by, trigger_role, lat, lng, address, status)
		VALUES ($1, $2, $3, $4, $5, $6, $7)
		RETURNING id, created_at`
	return r.pool.QueryRow(ctx, q,
		s.OrderID, s.TriggerBy, s.TriggerRole, s.Lat, s.Lng, s.Address, s.Status,
	).Scan(&s.ID, &s.CreatedAt)
}

// LatestByOrder 按 created_at DESC 取最近一条；订单无 SOS 时返回 (nil, nil)。
func (r *SosRepo) LatestByOrder(ctx context.Context, orderID int64) (*Record, error) {
	q := baseSelect + ` WHERE order_id = $1 ORDER BY created_at DESC LIMIT 1`
	row := r.pool.QueryRow(ctx, q, orderID)
	s, err := scanOne(row)
	if err != nil {
		if errors.Is(err, pgx.ErrNoRows) {
			return nil, nil
		}
		return nil, err
	}
	return s, nil
}

// Resolve 把 sos 状态改为 resolved。
func (r *SosRepo) Resolve(ctx context.Context, id, adminID int64, note string) error {
	const q = `
		UPDATE sos_records
		   SET status = 'resolved', resolved_at = NOW(), resolved_by = $2, resolution_note = $3
		 WHERE id = $1`
	tag, err := r.pool.Exec(ctx, q, id, adminID, note)
	if err != nil {
		return fmt.Errorf("resolve sos: %w", err)
	}
	if tag.RowsAffected() == 0 {
		return ErrSosNotFound
	}
	return nil
}

// Dismiss 把 sos 状态改为 dismissed（误触）。
func (r *SosRepo) Dismiss(ctx context.Context, id, adminID int64, note string) error {
	return r.Resolve(ctx, id, adminID, note) // 同样的 SQL（Dismiss 用 status='dismissed'）
}

// ListPending 返回 status='raised' 的最新 N 条（admin dashboard 队列）。
func (r *SosRepo) ListPending(ctx context.Context, limit int) ([]*Record, error) {
	q := baseSelect + ` WHERE status = 'raised' ORDER BY created_at DESC LIMIT $1`
	rows, err := r.pool.Query(ctx, q, limit)
	if err != nil {
		return nil, fmt.Errorf("list pending sos: %w", err)
	}
	defer rows.Close()
	out := make([]*Record, 0, limit)
	for rows.Next() {
		s, err := scanRow(rows)
		if err != nil {
			return nil, err
		}
		out = append(out, s)
	}
	return out, rows.Err()
}

// scanOne 把单行扫描为 *Record。
func scanOne(row pgx.Row) (*Record, error) {
	s := &Record{}
	if err := row.Scan(
		&s.ID, &s.OrderID, &s.TriggerBy, &s.TriggerRole, &s.Lat, &s.Lng, &s.Address,
		&s.Status, &s.CreatedAt, &s.ResolvedAt, &s.ResolvedBy, &s.ResolutionNote,
	); err != nil {
		return nil, err
	}
	return s, nil
}

// scanRow 把 pgx.Rows 的一行扫描为 *Record。
func scanRow(rows pgx.Rows) (*Record, error) {
	s := &Record{}
	if err := rows.Scan(
		&s.ID, &s.OrderID, &s.TriggerBy, &s.TriggerRole, &s.Lat, &s.Lng, &s.Address,
		&s.Status, &s.CreatedAt, &s.ResolvedAt, &s.ResolvedBy, &s.ResolutionNote,
	); err != nil {
		return nil, err
	}
	return s, nil
}

		return fmt.Errorf("marshal %s: %w", topic, err)

		Topic: topic, Key: []byte(key), Value: data, Time: time.Now(),
	})
}

		httpx.Fail(c, errs.New(errs.CodeParamInvalid, "invalid order id"))
		return

		return

		httpx.Fail(c, errs.New(errs.CodeParamInvalid, "trigger_role must be patient|escort"))
		return

	}
	pending, err := h.svc.ListPending(c.Request.Context(), limit)
	if err != nil {
		httpx.Fail(c, err)
		return
	}
	httpx.OK[[]map[string]any](c, pending)
}

// RegisterRoutes 挂路由。
func (h *Handler) RegisterRoutes(r gin.IRouter, authMW gin.HandlerFunc) {
	v1 := r.Group("/api/v1")
	if authMW != nil {
		v1.Use(authMW)
	}
	v1.POST("/orders/:id/sos", h.RaiseHandler)

	admin := v1.Group("/admin")
	if authMW != nil {
		admin.Use(authMW)
	}
	admin.POST("/sos/:id/resolve", h.ResolveHandler)
	admin.POST("/sos/:id/dismiss", h.DismissHandler)
	admin.GET("/sos/pending", h.ListPendingHandler)
}
```

// Package server 启动 sos-service HTTP server。
package server

import (
	"context"
	"net/http"
	"time"

	"go.uber.org/zap"

	"github.com/growdu/doctors/shared/logger"
)

// Server 是 HTTP server 包装。
type Server struct {
	addr string
	srv  *http.Server
}

// New 构造 server。
func New(addr string, h http.Handler) *Server {
	return &Server{
		addr: addr,
		srv: &http.Server{
			Addr:              addr,
			Handler:           h,
			ReadHeaderTimeout: 5 * time.Second,
		},
	}
}

// Run 启动 + 优雅停机。
func (s *Server) Run(ctx context.Context) error {
	errCh := make(chan error, 1)
	go func() {
		if err := s.srv.ListenAndServe(); err != nil && err != http.ErrServerClosed {
			errCh <- err
			return
		}
		errCh <- nil
	}()
	logger.L().Info("sos-service starting", zap.String("addr", s.addr))

	select {
	case <-ctx.Done():
		shutdownCtx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
		defer cancel()
		return s.srv.Shutdown(shutdownCtx)
	case err := <-errCh:
		return err
	}
}
```

// sos-service 入口。
package main

import (
	"context"

	"go.uber.org/zap/zapcore"

	"github.com/growdu/doctors/services/sos/internal/service"
	"github.com/growdu/doctors/shared/config"
	"github.com/growdu/doctors/shared/logger"
)

func main() {
	cfg, err := config.Load("sos")
	if err != nil {
		log.Fatalf("load config: %v", err)
	}
	logger.SetLevel(parseLevel(cfg.Logging.Level))
	defer func() { _ = logger.L().Sync() }()

	// pool 留 nil：业务调用会 panic；smoke 不走业务路径。
	sosRepo := repo.NewSosRepo(nil)

	var pub events.Publisher
	if len(cfg.Kafka.Brokers) > 0 {
		pub = events.NewKafkaPublisher(cfg.Kafka.Brokers)
	} else {
		pub = &events.NopPublisher{}
	}
	defer func() { _ = pub.Close() }()

	svc := service.New(sosRepo, nil, pub)
	h := handler.New(svc)
	srv := server.New(cfg.HTTP.Addr, router.New(h, cfg.Auth.JWTSecret))

	ctx, stop := signal.NotifyContext(context.Background(), os.Interrupt, syscall.SIGTERM)
	defer stop()

	logger.L().Info("sos-service starting", zap.String("addr", cfg.HTTP.Addr))
	if err := srv.Run(ctx); err != nil {
		logger.L().Error("sos-service exited", zap.Error(err))
		os.Exit(1)
	}
	logger.L().Info("sos-service stopped")
}

func parseLevel(s string) zapcore.Level {
	switch s {
	case "debug":
		return zapcore.DebugLevel
	case "warn":
		return zapcore.WarnLevel
	case "error":
		return zapcore.ErrorLevel
	default:
		return zapcore.InfoLevel
	}
}
